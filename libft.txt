static int	ft_strlen_atoi(const char *str)
{
	int	i;

	i = 0;
	while (*str == ' ' || (*str >= 9 && *str <= 13) || *str == '+'
		|| *str == '-'
		|| *str == '0')
		str++;
	while (*str >= '0' && *str <= '9')
	{
		str++;
		i++;
	}
	return (i);
}

static int	ft_recursive_power(int nb, int power)
{
	if (power < 0)
		return (0);
	if (power == 0)
		return (1);
	if (power == 1)
		return (nb);
	return (nb * ft_recursive_power(nb, (power - 1)));
}

const char	*skip(const char *str)
{
	int	minus_bool;
	int	nb;
	int	power;

	minus_bool = 0;
	power = ft_strlen_atoi(str);
	nb = 0;
	while (*str == ' ' || (*str >= 9 && *str <= 13))
		str++;
	if (*str == '-')
	{
		minus_bool = 1;
		str++;
	}
	else if (*str == '+')
		str++;
	while (*str == '0')
		str++;
	return (str);
}

int	ft_atoi(const char *str)
{
	str = skip(str);
	while (*str >= '0' && *str <= '9')
	{
		nb += (*str - '0') * ft_recursive_power(10, power - 1);
		power--;
		str++;
	}
	if (minus_bool == 0)
		return (nb);
	else
		return (-nb);
}

void	*ft_calloc(size_t n, size_t v_type)
{
	unsigned char	*ptr;
	size_t			i;
	size_t			test_size;

	test_size = v_type * n;
	if (n != 0 && test_size / n != v_type)
		return (0);
	ptr = malloc(v_type * n);
	if (ptr == 0)
		return (0);
	i = 0;
	while (i < n * v_type)
	{
		ptr[i] = '\0';
		i++;
	}
	return (ptr);
}

static size_t	int_len(int n)
{
	size_t	len;

	len = 0;
	if (n == 0)
		return (1);
	while (n != 0)
	{
		n = n / 10;
		len++;
	}
	return (len);
}

static char	*putnbr_itoa_pos(int n, char *ascii)
{
	size_t	i;

	i = int_len(n);
	ascii[i] = '\0';
	while (n != 0)
	{
		i--;
		ascii[i] = n % 10 + '0';
		n = n / 10;
	}
	return (ascii);
}

static char	*putnbr_itoa_neg(int n, char *ascii)
{
	size_t	i;

	if (n == -2147483648)
		ft_strlcpy(ascii, "-2147483648", 12);
	else
	{
		n = -n;
		i = int_len(n) + 1;
		ascii[i] = '\0';
		while (n != 0)
		{
			i--;
			ascii[i] = n % 10 + '0';
			n = n / 10;
		}
		i--;
		ascii[i] = '-';
	}
	return (ascii);
}

char	*ft_itoa(int n)
{
	char	*ascii;

	if (n < 0)
	{
		ascii = malloc(sizeof(char) * (int_len(n) + 2));
		if (ascii == 0)
			return (0);
		ascii = putnbr_itoa_neg(n, ascii);
	}
	else if (n == 0)
	{
		ascii = malloc(sizeof(char) * (int_len(n) + 1));
		if (ascii == 0)
			return (0);
		ascii[1] = '\0';
		ascii[0] = '0';
	}
	else
	{
		ascii = malloc(sizeof(char) * (int_len(n) + 1));
		if (ascii == 0)
			return (0);
		ascii = putnbr_itoa_pos(n, ascii);
	}
	return (ascii);
}

void	ft_lstadd_back(t_list **lst, t_list *new)
{
	t_list	*tmp;

	if (lst == 0 || new == 0)
		return ;
	if (*lst == 0)
	{
		*lst = new;
		new->next = 0;
		return ;
	}
	tmp = *lst;
	while (tmp->next != 0)
		tmp = tmp->next;
	tmp->next = new;
	new->next = 0;
	return ;
}

t_list	*ft_lstmap(t_list *lst, void *(*f)(void *), void (*del)(void *))
{
	t_list	*newlst;
	t_list	*const_newlst;
	t_list	*tmp;
	t_list	*lst_tmp;

	if (lst == 0 || f == 0 || del == 0)
		return (0);
	newlst = 0;
	const_newlst = newlst;
	lst_tmp = lst;
	while (lst_tmp != 0)
	{
		tmp = ft_lstnew(f(lst_tmp->content));
		if (tmp == 0)
		{
			ft_lstclear(&const_newlst, del);
			return (0);
		}
		ft_lstadd_back(&newlst, tmp);
		lst_tmp = lst_tmp->next;
	}
	return (const_newlst);
}

void	*ft_memchr(const void *s, int c, size_t n)
{
	size_t			i;
	unsigned char	*edit_s;

	i = 0;
	edit_s = (unsigned char *)s;
	while (i < n)
	{

#include "libft.h"

int	ft_memcmp(const void *s1, const void *s2, size_t n)
{
	size_t				i;
	const unsigned char	*sa;
	const unsigned char	*sb;

	i = 0;
	sa = s1;
	sb = s2;
	while (i < n)
	{
		if (*sa != *sb)
			return (*sa - *sb);
		sa++;
		sb++;
		i++;
	}
	return (0);
}

void	*ft_memcpy(void *dest, const void *src, size_t n)
{
	size_t				i;
	const unsigned char	*edit_src;
	unsigned char		*edit_dest;

	i = 0;
	edit_src = src;
	edit_dest = dest;
	while (i < n)
	{
		edit_dest[i] = edit_src[i];
		i++;
	}
	return (dest);
}

void	*ft_memmove(void *dest, const void *src, size_t n)
{
	unsigned char		*edit_dest;
	const unsigned char	*edit_src;

	edit_dest = dest;
	edit_src = src;
	if (dest > src)
	{
		while (n > 0)
		{
			edit_dest[n - 1] = edit_src[n - 1];
			n--;
		}
	}
	else
		ft_memcpy(dest, src, n);
	return (dest);
}

#include "libft.h"

static int	is_sep(char c, char charset)
{
	if (charset == c)
		return (1);
	return (0);
}

static int	strlen_split(char const *str, char charset, int count)
{
	while (*str != '\0')
	{
		while (*str != '\0' && is_sep(*str, charset) == 1)
			str++;
		if (*str != '\0')
			count++;
		while (*str != '\0' && is_sep(*str, charset) == 0)
			str++;
	}
	return (count);
}

static char	*copy_word(char const *str, char charset, int len, int i)
{
	char	*word;

	while (str[len] != '\0' && is_sep(str[len], charset) == 0)
		len++;
	word = malloc(sizeof(char) * (len + 1));
	if (word == 0)
		return (0);
	while (i < len)
	{
		word[i] = str[i];
		i++;
	}
	word[i] = '\0';
	return (word);
}

static void	clear(char **t, char **start)
{
	if (t == 0)
	{
		while (t >= start)
		{
			free(t);
			t--;
		}
	}
	return ;
}

char	**ft_split(char const *s, char c)
{
	char	**split_str;
	char	**const_split;

	if (s == 0)
		return (0);
	split_str = malloc(sizeof(char *) * (strlen_split(s, c, 0) + 1));
	if (split_str == 0)
		return (0);
	const_split = split_str;
	while (*s != '\0')
	{
		while (*s != '\0' && is_sep(*s, c) == 1)
			s++;
		if (*s != '\0')
		{
			*split_str = copy_word(s, c, 0, 0);
			clear(split_str, const_split);
			split_str++;
		}
		while (*s != '\0' && is_sep(*s, c) == 0)
			s++;
	}
	*split_str = 0;
	return (const_split);
}

char	*ft_strjoin(char const *s1, char const *s2)
{
	char	*str;
	size_t	i;
	size_t	s2_len;

	s2_len = ft_strlen(s2);
	str = malloc(sizeof(char) * (ft_strlen(s1) + ft_strlen(s2) + 1));
	if (str == 0)
		return (0);
	i = 0;
	while (i < ft_strlen(s1))
	{
		str[i] = s1[i];
		i++;
	}
	while (i < ft_strlen(s1) + s2_len)
	{
		str[i] = *s2;
		s2++;
		i++;
	}
	str[i] = '\0';
	return (str);
}

{
	size_t	i;
	size_t	j;
	size_t	dest_size;

	i = 0;
	j = 0;
	dest_size = ft_strlen(dest);
	if (dstsize <= dest_size)
		return (dstsize + ft_strlen(src));
	while (dest[i] != '\0')
		i++;
	while (j < dstsize - dest_size - 1 && src[j] != '\0')
	{
		dest[i] = src[j];
		i++;
		j++;
	}
	dest[i] = '\0';
	return (dest_size + ft_strlen(src));
}

{

	i = 0;
	j = 0;
	if (*little == '\0')
		return ((char *)big);
	while (big[i] != '\0' && i < len)
	{
		if (big[i] == little[j])
		{
			if (little[j + 1] == '\0')
				return ((char *)&big[i - j]);
			j++;
		}
		else
		{
			i -= j;
			j = 0;
		}
		i++;
	}
	return (0);
}

static int	is_set(char c, char const *set)
{
	size_t	i;

	i = 0;
	while (set[i] != '\0')
	{
		if (set[i] == c)
			return (1);
		i++;
	}
	return (0);
}

static size_t	strtrim_len(char const *s1, char const *set)
{
	size_t	i;
	size_t	j;

	i = 0;
	j = ft_strlen(s1) - 1;
	while (is_set(s1[i], set) == 1)
		i++;
	if (i != ft_strlen(s1))
	{
		while (is_set(s1[j], set) == 1 && j > 0)
		{
			i++;
			j--;
		}
	}
	return (ft_strlen(s1) - i);
}

char	*ft_strtrim(char const *s1, char const *set)
{
	size_t	i;
	char	*new_str;
	char	*const_new_str;
	size_t	trim_len;

	i = 0;
	trim_len = strtrim_len(s1, set);
	new_str = malloc(sizeof(char) * (trim_len + 1));
	if (new_str == 0)
		return (0);
	const_new_str = new_str;
	while (is_set(*s1, set) == 1)
		s1++;
	while (i < trim_len)
	{
		*new_str = *s1;
		s1++;
		i++;
		new_str++;
	}
	*new_str = '\0';
	return (const_new_str);
}

	size_t	i;
	char	*substring;

	i = 0;
	if (ft_strlen(s) < start)
		return (ft_strdup("\0"));
	substring = malloc(sizeof(char) * (ft_strlen(s) - start + 1));
	if (substring == 0)
		return (0);
	while (start < ft_strlen(s))
	{
		substring[i] = s[start];
		start++;
		i++;
	}
	substring[i] = '\0';
	return (substring);
}

static char	*ft_substr_underflow(char const *s, unsigned int start, size_t len)
{
	size_t	i;
	char	*substring;

	i = 0;
	substring = malloc(sizeof(char) * (len + 1));
	if (substring == 0)
		return (0);
	while (i < len)
	{
		substring[i] = s[start];
		start++;
		i++;
	}
	substring[i] = '\0';
	return (substring);
}

char	*ft_substr(char const *s, unsigned int start, size_t len)
{
	if (start + len > ft_strlen(s))
		return (ft_substr_overflow(s, start));
	else
		return (ft_substr_underflow(s, start, len));
}

int	ft_tolower(int c)
{
	if (c >= 'A' && c <= 'Z')
		return (c + 32);
	else
		return (c);
}

int	ft_toupper(int c)
{
	if (c >= 'a' && c <= 'z')
		return (c - 32);
	else
		return (c);
}
